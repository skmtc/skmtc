Skmtc offers extensive configuration options to customize code generation for your specific needs.

## Configuration Files

### File Structure

Skmtc uses a `.skmtc` directory in your project root:

<FileTree>
  <FileTree.Folder name=".skmtc" defaultOpen>
    <FileTree.File name="config.json" />
    <FileTree.Folder name="generators">
      <FileTree.File name="typescript.json" />
      <FileTree.File name="tanstack-query.json" />
      <FileTree.File name="custom-generator.js" />
    </FileTree.Folder>
    <FileTree.Folder name="templates">
      <FileTree.File name="model.hbs" />
      <FileTree.File name="operation.hbs" />
    </FileTree.Folder>
    <FileTree.Folder name="cache">
      <FileTree.File name="manifest.json" />
    </FileTree.Folder>
  </FileTree.Folder>
</FileTree>

### Main Configuration

The `.skmtc/config.json` file:

```json
{
  "$schema": "https://skmtc.dev/schemas/config.json",
  "version": "1.0.0",
  "source": "./openapi.json",
  "output": {
    "baseDir": "./src/generated",
    "clean": true,
    "structure": "nested"
  },
  "generators": [
    "@skmtc/gen-typescript",
    "@skmtc/gen-zod",
    "@skmtc/gen-tanstack-query"
  ],
  "formatting": {
    "prettier": true,
    "prettierConfig": "./.prettierrc"
  }
}
```

## Core Settings

### Source Configuration

Configure OpenAPI source location:

```json
{
  "source": {
    // Single source
    "url": "https://api.example.com/openapi.json",

    // Or local file
    "file": "./specs/openapi.yaml",

    // With authentication
    "url": "https://api.example.com/openapi.json",
    "headers": {
      "Authorization": "Bearer ${SKMTC_API_TOKEN}"
    },

    // Multiple sources
    "sources": [
      { "name": "users", "file": "./specs/users.yaml" },
      { "name": "products", "url": "https://api.example.com/products/spec" }
    ]
  }
}
```

### Output Configuration

Control generated file structure:

```json
{
  "output": {
    "baseDir": "./src/generated",
    "structure": "nested",  // "flat" | "nested" | "domain"
    "clean": false,         // Clean directory before generation
    "gitignore": true,      // Create .gitignore

    // Custom structure mapping
    "paths": {
      "types": "models",
      "hooks": "queries",
      "schemas": "validation"
    },

    // File naming
    "naming": {
      "case": "kebab",     // "camel" | "pascal" | "kebab" | "snake"
      "suffix": {
        "types": ".types",
        "hooks": ".hooks",
        "schemas": ".schema"
      }
    }
  }
}
```

### Generator Configuration

Configure individual generators:

```json
{
  "generators": {
    "@skmtc/gen-typescript": {
      "enumStyle": "union",        // "union" | "enum" | "const"
      "dateType": "Date",          // "Date" | "string"
      "optionalStyle": "questionMark", // "questionMark" | "undefined"
      "readonlyProps": false,
      "exactTypes": true,
      "exportType": "named"        // "named" | "default" | "namespace"
    },

    "@skmtc/gen-zod": {
      "strict": true,
      "coerce": {
        "date": true,
        "number": false
      },
      "errorMessages": true,
      "transforms": ["trim", "toLowerCase"]
    },

    "@skmtc/gen-tanstack-query": {
      "version": 5,
      "suspense": false,
      "optimisticUpdates": true,
      "customClient": "./api-client",
      "queryKeyStyle": "array"    // "array" | "object" | "factory"
    }
  }
}
```

## Advanced Configuration

### Schema Filters

Control which schemas to process:

```json
{
  "filters": {
    // Include only specific schemas
    "include": {
      "models": ["User", "Product", "Order"],
      "operations": ["getUser", "createProduct"],
      "tags": ["public", "v2"]
    },

    // Exclude specific schemas
    "exclude": {
      "models": ["Internal*", "Debug*"],
      "operations": ["deprecated*"],
      "tags": ["internal", "experimental"]
    },

    // Regex patterns
    "patterns": {
      "models": "^Public",
      "operations": "^(get|list)"
    }
  }
}
```

### Naming Conventions

Customize naming throughout:

```json
{
  "naming": {
    // Global naming rules
    "global": {
      "case": "camelCase",
      "removePrefix": ["API", "DTO"],
      "removeSuffix": ["Model", "Schema"]
    },

    // Per-generator naming
    "generators": {
      "@skmtc/gen-typescript": {
        "models": "PascalCase",
        "properties": "camelCase",
        "enums": "UPPER_SNAKE_CASE"
      },
      "@skmtc/gen-hooks": {
        "hooks": "use{Operation}",
        "queryKeys": "{operation}Key"
      }
    },

    // Custom transformers
    "transforms": [
      { "from": "UUID", "to": "Id" },
      { "from": "URL", "to": "Url" }
    ]
  }
}
```

### Import Management

Configure import generation:

```json
{
  "imports": {
    "style": "named",           // "named" | "namespace" | "default"
    "extensions": true,         // Add .js extensions
    "aliases": {
      "@types": "./types",
      "@hooks": "./hooks",
      "@utils": "./utils"
    },
    "external": {
      "zod": "z",              // Import { z } from 'zod'
      "react-query": "rq"      // Import * as rq from 'react-query'
    },
    "sorting": "alphabetical", // "alphabetical" | "grouped" | "natural"
    "grouping": [
      { "pattern": "react", "group": "react" },
      { "pattern": "@/", "group": "internal" },
      { "pattern": ".", "group": "relative" }
    ]
  }
}
```

```json

      "strictNullChecks": true,
      "exactOptionalPropertyTypes": true

  }

```

## Environment Variables

Use environment variables in configuration:

```json
{
  "source": {
    "url": "${API_BASE_URL}/openapi.json",
    "headers": {
      "Authorization": "Bearer ${API_TOKEN}",
      "X-API-Version": "${API_VERSION:-v1}"
    }
  },
  "output": {
    "baseDir": "${OUTPUT_DIR:-./generated}"
  }
}
```

Load from `.env` file:

```bash
# .env
API_BASE_URL=https://api.example.com
API_TOKEN=secret-token
OUTPUT_DIR=./src/api
```

## Configuration Profiles

Use different configs for different environments:

<Tabs items={['Development', 'Production', 'Testing']}>
<Tab value="Development">
```json
// .skmtc/config.development.json
{
  "extends": "./config.json",
  "source": {
    "url": "http://localhost:3000/api/openapi.json"
  },
  "generators": {
    "@skmtc/gen-msw": {
      "enabled": true
    }
  },
  "formatting": {
    "prettier": false  // Faster in development
  }
}
```
</Tab>
<Tab value="Production">
```json
// .skmtc/config.production.json
{
  "extends": "./config.json",
  "source": {
    "url": "https://api.example.com/openapi.json"
  },
  "output": {
    "clean": true,
    "minify": true
  },
  "validation": {
    "strict": true
  }
}
```
</Tab>
<Tab value="Testing">
```json
// .skmtc/config.test.json
{
  "extends": "./config.json",
  "generators": [
    "@skmtc/gen-typescript",
    "@skmtc/gen-jest",
    "@skmtc/gen-msw"
  ],
  "output": {
    "baseDir": "./__generated_test__"
  }
}
```
</Tab>
</Tabs>

Use profiles:

```bash
# Use specific profile
skmtc generate --config .skmtc/config.production.json

# Or with environment variable
SKMTC_PROFILE=production skmtc generate
```

## Validation Settings

Configure schema validation:

```json
{
  "validation": {
    "mode": "strict",          // "strict" | "warning" | "lenient"
    "openapi": {
      "strict": true,
      "allowExtensions": true,
      "validateExamples": true,
      "checkRefs": true
    },
    "rules": {
      "requireDescriptions": "warning",
      "requireExamples": "off",
      "uniqueOperationIds": "error",
      "noEmptySchemas": "warning"
    },
    "custom": [
      {
        "rule": "operation-id-format",
        "pattern": "^[a-z][a-zA-Z0-9]*$",
        "level": "error"
      }
    ]
  }
}
```

## Plugin System

Extend Skmtc with plugins:

```json
{
  "plugins": [
    {
      "name": "custom-validator",
      "path": "./plugins/validator.js",
      "options": {
        "strict": true
      }
    },
    {
      "name": "@skmtc/plugin-analytics",
      "options": {
        "trackUsage": true
      }
    }
  ],
  "hooks": {
    "beforeParse": ["./hooks/pre-parse.js"],
    "afterGenerate": ["./hooks/post-generate.js"],
    "beforeWrite": ["./hooks/pre-write.js"]
  }
}
```

## Schema Extensions

Add custom extensions to OpenAPI:

```json
{
  "extensions": {
    "x-skmtc": {
      "generator-config": {
        "@skmtc/gen-typescript": {
          "enumStyle": "const"
        }
      }
    },
    "x-custom": {
      "domain": "users",
      "version": "v2"
    }
  }
}
```

Use in OpenAPI spec:

```yaml
User:
  type: object
  x-skmtc:
    readonly: true
    customType: "UserModel"
  x-custom:
    cacheable: true
```

## Configuration Schema

Enable IDE autocomplete:

```json
{
  "$schema": "https://skmtc.dev/schemas/config.json",
  // Your config with full IntelliSense
}
```

Or use TypeScript for configuration:

```typescript
// .skmtc/config.ts
import { defineConfig } from '@skmtc/cli';

export default defineConfig({
  source: './openapi.json',
  output: {
    baseDir: './src/generated',
  },
  generators: [
    '@skmtc/gen-typescript',
    {
      name: '@skmtc/gen-tanstack-query',
      config: {
        version: 5,
        suspense: true
      }
    }
  ]
});
```

## Best Practices

<Callout type="tip">
**Configuration Tips:**

1. **Start simple** - Use defaults and add configuration as needed
2. **Use profiles** - Separate dev/prod configurations
3. **Version control** - Commit `.skmtc/config.json` but not cache
4. **Environment variables** - Keep secrets out of config files
5. **Validate early** - Use strict validation in CI/CD
</Callout>

## Troubleshooting

### Configuration not loading

```bash
# Check config location
skmtc config --show

# Validate configuration
skmtc config --validate

# Debug config loading
SKMTC_DEBUG=config skmtc generate
```

### Generator conflicts

```json
{
  "generators": {
    "@skmtc/gen-typescript": {
      "priority": 1  // Run first
    },
    "@skmtc/gen-zod": {
      "priority": 2,
      "dependsOn": ["@skmtc/gen-typescript"]
    }
  }
}
```
